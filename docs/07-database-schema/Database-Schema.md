## 1. Vue d'Ensemble

Lumina utilise **WatermelonDB** comme ORM local SQLite. Chaque modèle WatermelonDB correspond à une table PostgreSQL côté serveur InsForge. Les modèles locaux utilisent les colonnes obligatoires WatermelonDB (`id`, `_synced`, `createdAt`, `updatedAt`).

```
┌─────────────────┐     ┌──────────────────────┐
│  App React Native │────▶│  WatermelonDB (SQLite) │
│  (presentation)   │     │  .synced = clean     │
└─────────────────┘     │  .pending = dirty      │
                        └──────────┬───────────┘
                                   │ sync()
                                   ▼
                        ┌──────────────────────┐
                        │  InsForge / PostgreSQL │
                        │  (source of truth)     │
                        └──────────────────────┘
```

## 2. Modèles WatermelonDB

### 2.1 Transaction (Core Finance)

```typescript
import { Table, props, column, children } from '@nozbe/watermelondb/decorators'

@Table('transactions')
export class Transaction extends Model {
  static table = 'transactions' as const

  // === COLONNES WATERMELONDB OBLIGATOIRES ===
  id            = prop()               // PK auto-generated by WatermelonDB
  _synced       = column('synced', 0)  // 0=pending, 1=synced, 2=conflict
  createdAt     = column('created_at')
  updatedAt     = column('updated_at')

  // === CHAMPS MÉTIER ===
  orgId         = column('org_id')
  type          = column('type')           // 'income' | 'expense' | 'transfer'
  amount        = column('amount')         // decimal stored as number (x100)
  currency      = column('currency')       // 'CDF', 'USD', etc.
  categoryId    = column('category_id')    // FK → categories.id
  description   = column('description')    // nullable
  receiptPath   = column('receipt_path')   // local file path or null
  date          = column('transaction_date')
  status        = column('status')         // 'draft' | 'pending' | 'approved' | 'rejected'
  createdBy     = column('created_by')     // user UUID
  approvedBy    = column('approved_by')    // nullable
  approvedAt    = column('approved_at')    // nullable
  version       = column('version', 1)     // optimistic lock version
}
```

**IndexSQLite :**
```sql
CREATE INDEX idx_transactions_org ON transactions(org_id);
CREATE INDEX idx_transactions_status ON transactions(status);
CREATE INDEX idx_transactions_date ON transactions(transaction_date);
CREATE INDEX idx_transactions_synced ON transactions(synced);
```

### 2.2 Category (Vocabulaire Financier)

```typescript
@Table('categories')
export class Category extends Model {
  static table = 'categories' as const

  id        = prop()
  _synced   = column('synced', 1)   // always synced (lookup table)
  createdAt = column('created_at')
  updatedAt = column('updated_at')

  orgId         = column('org_id')
  name          = column('name')         // i18n key → vocabulary resolves label
  nameFr        = column('name_fr')      // French display name (fallback)
  parentCategoryId = column('parent_category_id', null)  // self-referencing FK
  categoryType  = column('category_type') // 'income' | 'expense' | 'asset' | 'liability' | 'equity'
  isImmutable   = column('is_immutable', false)
}
```

**Règle critique :** Les catégories avec `is_immutable=true` ne peuvent pas être supprimées ni modifiées. Seuls les admins peuvent créer des catégories.

## 3. Schéma PostgreSQL (Serveur)

Chaque modèle local a un équivalent serveur avec **Row-Level Security (RLS)** :

```sql
-- Table transactions (serveur)
CREATE TABLE organizations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    type TEXT NOT NULL, -- church, school, ngo, company
    settings JSONB DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id UUID REFERENCES organizations(id) ON DELETE CASCADE,
    email TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    role TEXT NOT NULL DEFAULT 'member',
    first_name TEXT,
    last_name TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- CREATE INDEX idx_users_org ON users(org_id);
-- ALTER TABLE users ENABLE ROW LEVEL SECURITY;
-- CREATE POLICY org_isolation ON users USING (org_id = current_setting('app.current_org_id')::uuid);

CREATE TABLE transactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
    type TEXT NOT NULL CHECK (type IN ('income', 'expense', 'transfer')),
    amount BIGINT NOT NULL CHECK (amount > 0),  -- stored in cents/fractions
    currency TEXT NOT NULL DEFAULT 'CDF',
    category_id UUID REFERENCES categories(id),
    description TEXT,
    receipt_path TEXT,
    transaction_date DATE NOT NULL CHECK (transaction_date <= CURRENT_DATE),
    status TEXT NOT NULL DEFAULT 'draft' CHECK (status IN ('draft', 'pending', 'approved', 'rejected', 'archived')),
    created_by UUID REFERENCES users(id),
    approved_by UUID REFERENCES users(id),
    approved_at TIMESTAMPTZ,
    version INTEGER NOT NULL DEFAULT 1,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- CRITICAL: Indexes pour perf
CREATE INDEX idx_transactions_org ON transactions(org_id);
CREATE INDEX idx_transactions_status ON transactions(status);
CREATE INDEX idx_transactions_date ON transactions(transaction_date);
-- Composite pour les requêtes de bilan fréquentes
CREATE INDEX idx_transactions_org_status_date ON transactions(org_id, status, transaction_date);

ALTER TABLE transactions ENABLE ROW LEVEL SECURITY;
CREATE POLENTITY org_isolation ON transactions USING (org_id = current_setting('app.current_org_id')::uuid);

CREATE TABLE categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    name_fr TEXT,
    parent_category_id UUID REFERENCES categories(id) ON DELETE SET NULL,
    category_type TEXT NOT NULL CHECK (category_type IN ('income', 'expense', 'asset', 'liability', 'equity')),
    is_immutable BOOLEAN NOT NULL DEFAULT false,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);
```

## 5. Règles Critiques pour les Agents IA

| Règle | Détail |
|---|---|
| **DS-R01** | Toujours utiliser `@prop()` et `@column()` — jamais de champs privés non décorés |
| **DS-R02** | Colonnes WatermelonDB obligatoire : `id`, `_synced`, `createdAt`, `updatedAt` |
| **DS-R03** | Les nombres décimaux doivent être stockés en **cents/fractions** (amount × 100) pour éviter les erreurs floating-point |
| **DS-R04** | Les relations parents/enfants utilisent des foreign keys simples (WatermelonDB gère les `.relation()`) |
| **DS-R05** | `pending_operations` ne doit JAMAIS être purgé sans confirmation explicite |
| **DS-R06** | Toute modification de schéma SQLite doit inclure une migration versionnée |
| **DS-R07** | Les timestamps sont toujours en UTC, jamais en local time |

## 6. Requête Bilan (Exemple de Calcul)

```typescript
// src/features/finance/bilan-calculator.ts
import { Q } from '@nozbe/watermelondb'

async function calculateBilan(db: Database, orgId: string, year: number, month?: number) {
  // Seules les transactions SYNCED (non en attente) participent au calcul
  const query = month
    ? db.get('transactions').query(
        Q.where('org_id', orgId),
        Q.where('synced', 1),
        Q.onDateFuzzyBetween('transaction_date', `${year}-${month}-01`, `${year}-${month}-${getLastDayOfMonth(year, month)}`),
        Q.not('status', 'rejected')
      )
    : db.get('transactions').query(
        Q.where('org_id', orgId),
        Q.where('synced', 1),
        Q.where('transaction_date', '>=', `${year}-01-01`),
        Q.where('transaction_date', '<=', `${year}-12-31`),
        Q.not('status', 'rejected')
      )

  const transactions = await query.fetch()
  
  const income = transactions
    .filter(t => t.type === 'income')
    .reduce((sum, t) => sum + t.amount, 0)
    
  const expense = transactions
    .filter(t => t.type === 'expense')
    .reduce((sum, t) => sum + t.amount, 0)

  return { income, expense, netResult: income - expense }
}
```

**Point critique :** La clause `Q.where('synced', 1)` exclut les transactions en cours de sync. C'est intentional — elles seront comptabilisées après confirmation serveur.
